cmake_minimum_required(VERSION 3.20)
project(ForgeDB VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FORGEDB_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
set(FORGEDB_SANITIZER "" CACHE STRING "Sanitizer: address, undefined, address,undefined, or thread")
find_package(Threads REQUIRED)
add_library(forgedb
  src/protocol/resp.cpp src/storage/store.cpp src/persistence/file.cpp
  src/persistence/aof.cpp src/commands/engine.cpp src/commands/session.cpp
  src/networking/socket.cpp src/networking/client.cpp src/server/server.cpp)
target_include_directories(forgedb PUBLIC include)
target_link_libraries(forgedb PUBLIC Threads::Threads)
if(WIN32)
  target_compile_definitions(forgedb PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
  target_link_libraries(forgedb PUBLIC ws2_32)
endif()
if(MSVC)
  target_compile_options(forgedb PUBLIC /W4 /permissive- /EHsc)
  if(FORGEDB_WARNINGS_AS_ERRORS)
    target_compile_options(forgedb PUBLIC /WX)
  endif()
  if(FORGEDB_SANITIZER STREQUAL "address")
    target_compile_options(forgedb PUBLIC /fsanitize=address /Zi)
    target_link_options(forgedb PUBLIC /INCREMENTAL:NO)
  elseif(FORGEDB_SANITIZER)
    message(FATAL_ERROR "MSVC supports only address sanitizer here")
  endif()
else()
  target_compile_options(forgedb PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(FORGEDB_WARNINGS_AS_ERRORS)
    target_compile_options(forgedb PUBLIC -Werror)
  endif()
  if(FORGEDB_SANITIZER)
    target_compile_options(forgedb PUBLIC -fsanitize=${FORGEDB_SANITIZER} -fno-omit-frame-pointer -g)
    target_link_options(forgedb PUBLIC -fsanitize=${FORGEDB_SANITIZER})
  endif()
endif()
add_executable(forgedb-server src/main.cpp)
add_executable(forgedb-cli cli/main.cpp)
add_executable(forgedb-benchmark benchmarks/main.cpp)
add_executable(forgedb-parser-benchmark benchmarks/parser.cpp)
foreach(app forgedb-server forgedb-cli forgedb-benchmark forgedb-parser-benchmark)
  target_link_libraries(${app} PRIVATE forgedb)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(forgedb-unit tests/unit/main.cpp)
  target_link_libraries(forgedb-unit PRIVATE forgedb)
  add_test(NAME unit COMMAND forgedb-unit)
  add_executable(forgedb-test-server src/main.cpp)
  target_compile_definitions(forgedb-test-server PRIVATE FORGEDB_TEST_DRIVER=1)
  target_link_libraries(forgedb-test-server PRIVATE forgedb)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/test_server.py $<TARGET_FILE:forgedb-server>)
  add_test(NAME recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/recovery/test_recovery.py $<TARGET_FILE:forgedb-server> $<TARGET_FILE:forgedb-test-server>)
  set_tests_properties(unit integration recovery PROPERTIES TIMEOUT 180)
endif()
install(TARGETS forgedb-server forgedb-cli forgedb-benchmark RUNTIME DESTINATION bin)
